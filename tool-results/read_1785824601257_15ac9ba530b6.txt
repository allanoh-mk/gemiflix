    29→    29→function getVolumeIcon(volume: number, isMuted: boolean): VolumeIcon {
    30→    30→  if (isMuted || volume === 0) return 'muted';

    35→    35→function formatTime(seconds: number): string {
    36→    36→  const h = Math.floor(seconds / 3600);
    37→    37→  const m = Math.floor((seconds % 3600) / 60);
    38→    38→  const s = Math.floor(seconds % 60);
    39→    39→  if (h > 0) {
    40→    40→    return `${h}:${m.toString().padStart(2, '0')}:${s.toString().padStart(2, '0')}`;
    41→    41→  }
    42→    42→  return `${m.toString().padStart(2, '0')}:${s.toString().padStart(2, '0')}`;
    43→    43→}

    45→    45→const PROGRESS_STORAGE_KEY = 'gemiflix-player-progress';
    46→    46→
    47→    47→function loadSavedProgress(movieId: string): number {
    48→    48→  if (typeof window === 'undefined') return 0;
    49→    49→  try {
    50→    50→    const raw = localStorage.getItem(PROGRESS_STORAGE_KEY);
    51→    51→    if (!raw) return 0;
    52→    52→    const map = JSON.parse(raw) as Record<string, number>;
    53→    53→    return map[movieId] ?? 0;
    54→    54→  } catch {
    55→    55→    return 0;
    56→    56→  }
    57→    57→}
    58→    58→
    59→    59→function saveProgressToStorage(movieId: string, time: number): void {
    60→    60→  if (typeof window === 'undefined') return;
    61→    61→  try {
    62→    62→    const raw = localStorage.getItem(PROGRESS_STORAGE_KEY);
    63→    63→    const map = raw ? (JSON.parse(raw) as Record<string, number>) : {};
    64→    64→    map[movieId] = time;
    65→    65→    localStorage.setItem(PROGRESS_STORAGE_KEY, JSON.stringify(map));
    66→    66→  } catch {
    67→    67→    // storage full or unavailable
    68→    68→  }
    69→    69→}

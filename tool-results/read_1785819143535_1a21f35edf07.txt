    26→function getVolumeIcon(volume: number, isMuted: boolean): VolumeIcon {
    27→  if (isMuted || volume === 0) return 'muted';

    29→  return 'high';

    32→function formatTime(seconds: number): string {
    33→  const h = Math.floor(seconds / 3600);
    34→  const m = Math.floor((seconds % 3600) / 60);
    35→  const s = Math.floor(seconds % 60);
    36→  if (h > 0) {
    37→    return `${h}:${m.toString().padStart(2, '0')}:${s.toString().padStart(2, '0')}`;
    38→  }
    39→  return `${m.toString().padStart(2, '0')}:${s.toString().padStart(2, '0')}`;
    40→}
    41→
    42→function formatRemainingTime(current: number, total: number): string {
    43→  const remaining = Math.max(0, total - current);
    44→  const negH = Math.floor(remaining / 3600);
    45→  const negM = Math.floor((remaining % 3600) / 60);
    46→  const negS = Math.floor(remaining % 60);
    47→  const totH = Math.floor(total / 3600);
    48→  const totM = Math.floor((total % 3600) / 60);
    49→  const totS = Math.floor(total % 60);
    50→  const fmtRemain = negH > 0
    51→    ? `-${negH}:${negM.toString().padStart(2, '0')}:${negS.toString().padStart(2, '0')}`
    52→    : `-${negM.toString().padStart(2, '0')}:${negS.toString().padStart(2, '0')}`;
    53→  const fmtTotal = totH > 0
    54→    ? `${totH}:${totM.toString().padStart(2, '0')}:${totS.toString().padStart(2, '0')}`
    55→    : `${totM.toString().padStart(2, '0')}:${totS.toString().padStart(2, '0')}`;
    56→  return `${fmtRemain} / ${fmtTotal}`;
    57→}

    59→const PROGRESS_STORAGE_KEY = 'gemiflix-player-progress';
    60→
    61→function loadSavedProgress(movieId: string): number {
    62→  if (typeof window === 'undefined') return 0;
    63→  try {
    64→    const raw = localStorage.getItem(PROGRESS_STORAGE_KEY);
    65→    if (!raw) return 0;
    66→    const map = JSON.parse(raw) as Record<string, number>;
    67→    return map[movieId] ?? 0;
    68→  } catch {
    69→    return 0;
    70→  }
    71→}
    72→
    73→function saveProgressToStorage(movieId: string, time: number): void {
    74→  if (typeof window === 'undefined') return;
    75→  try {
    76→    const raw = localStorage.getItem(PROGRESS_STORAGE_KEY);
    77→    const map = raw ? (JSON.parse(raw) as Record<string, number>) : {};
    78→    map[movieId] = time;
    79→    localStorage.setItem(PROGRESS_STORAGE_KEY, JSON.stringify(map));
    80→  } catch {
    81→    // storage full or unavailable
    82→  }
    83→}

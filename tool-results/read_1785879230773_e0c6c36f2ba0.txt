   397→/* Glassmorphic select */
   398→function GlassSelect({ value, onValueChange, options, className = '' }: {
   399→  value: string; onValueChange: (v: string) => void;
   400→  options: { value: string; label: string }[];
   401→  className?: string;
   402→}) {
   403→  return (
   404→    <select
   405→      value={value}
   406→      onChange={(e) => onValueChange(e.target.value)}
   407→      className={`bg-black/80 backdrop-blur-xl border border-[var(--glass-border)] rounded-lg px-3 py-1.5 text-sm text-[var(--foreground)] outline-none transition-colors focus:border-[var(--accent-current)]/50 ${className}`}
   408→    >
   409→      {options.map((opt) => (
   410→        <option key={opt.value} value={opt.value}>{opt.label}</option>
   411→      ))}
   412→    </select>
   413→  );
   414→}

   463→function ProfileTab({ settings, setSettings, onSave }: {
   464→  settings: UserSettingsData; setSettings: React.Dispatch<React.SetStateAction<UserSettingsData>>; onSave: () => void;
   465→}) {
   466→  const user = useAuthStore((s) => s.user);
   467→  const fileInputRef = useRef<HTMLInputElement>(null);
   468→  const [passwordError, setPasswordError] = useState('');
   469→
   470→  const handleAvatarUpload = (e: React.ChangeEvent<HTMLInputElement>) => {
   471→    const file = e.target.files?.[0];
   472→    if (!file) return;
   473→    const reader = new FileReader();
   474→    reader.onloadend = () => {
   475→      setSettings((prev) => ({ ...prev, avatar: reader.result as string }));
   476→    };
   477→    reader.readAsDataURL(file);
   478→  };
   479→
   480→  const handleChangePassword = () => {
   481→    if (!settings.currentPassword) {
   482→      setPasswordError('Current password is required');
   483→      return;
   484→    }
   485→    if (settings.newPassword.length < 8) {
   486→      setPasswordError('Password must be at least 8 characters');
   487→      return;
   488→    }
   489→    if (settings.newPassword !== settings.confirmPassword) {
   490→      setPasswordError('Passwords do not match');
   491→      return;
   492→    }
   493→    setPasswordError('');
   494→    toast.success('Password changed successfully');
   495→    setSettings((prev) => ({ ...prev, currentPassword: '', newPassword: '', confirmPassword: '' }));
   496→  };
   497→
   498→  return (
   499→    <div className="space-y-4">
   500→      <SettingCard>
   501→        <SectionHeader title="Profile Information" icon={<User className="w-4 h-4" />} />
   502→
   503→        {/* Display Name */}
   504→        <SettingRow label="Display Name" description="Your public display name">
   505→          <GlassInput
   506→            value={settings.displayName || user?.name || ''}
   507→            onChange={(v) => setSettings((p) => ({ ...p, displayName: v }))}
   508→            placeholder="Enter your name"

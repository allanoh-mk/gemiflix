    24→function getVolumeIcon(volume: number, isMuted: boolean): VolumeIcon {
    25→  if (isMuted || volume === 0) return 'muted';
    26→  if (volume < 0.5) return 'low';
    27→  return 'high';

    30→function formatTime(seconds: number): string {
    31→  const h = Math.floor(seconds / 3600);
    32→  const m = Math.floor((seconds % 3600) / 60);
    33→  const s = Math.floor(seconds % 60);
    34→  if (h > 0) {
    35→    return `${h}:${m.toString().padStart(2, '0')}:${s.toString().padStart(2, '0')}`;
    36→  }
    37→  return `${m.toString().padStart(2, '0')}:${s.toString().padStart(2, '0')}`;
    38→}
    39→
    40→function formatRemainingTime(current: number, total: number): string {
    41→  const remaining = Math.max(0, total - current);
    42→  const negH = Math.floor(remaining / 3600);
    43→  const negM = Math.floor((remaining % 3600) / 60);
    44→  const negS = Math.floor(remaining % 60);
    45→  const totH = Math.floor(total / 3600);
    46→  const totM = Math.floor((total % 3600) / 60);
    47→  const totS = Math.floor(total % 60);
    48→  const fmtRemain = negH > 0
    49→    ? `-${negH}:${negM.toString().padStart(2, '0')}:${negS.toString().padStart(2, '0')}`
    50→    : `-${negM.toString().padStart(2, '0')}:${negS.toString().padStart(2, '0')}`;
    51→  const fmtTotal = totH > 0
    52→    ? `${totH}:${totM.toString().padStart(2, '0')}:${totS.toString().padStart(2, '0')}`
    53→    : `${totM.toString().padStart(2, '0')}:${totS.toString().padStart(2, '0')}`;
    54→  return `${fmtRemain} / ${fmtTotal}`;
    55→}

    57→const PROGRESS_STORAGE_KEY = 'gemiflix-player-progress';
    58→
    59→function loadSavedProgress(movieId: string): number {
    60→  if (typeof window === 'undefined') return 0;
    61→  try {
    62→    const raw = localStorage.getItem(PROGRESS_STORAGE_KEY);
    63→    if (!raw) return 0;
    64→    const map = JSON.parse(raw) as Record<string, number>;
    65→    return map[movieId] ?? 0;
    66→  } catch {
    67→    return 0;
    68→  }
    69→}
    70→
    71→function saveProgressToStorage(movieId: string, time: number): void {
    72→  if (typeof window === 'undefined') return;
    73→  try {
    74→    const raw = localStorage.getItem(PROGRESS_STORAGE_KEY);
    75→    const map = raw ? (JSON.parse(raw) as Record<string, number>) : {};
    76→    map[movieId] = time;
    77→    localStorage.setItem(PROGRESS_STORAGE_KEY, JSON.stringify(map));
    78→  } catch {
    79→    // storage full or unavailable
    80→  }
    81→}

  1962→function AdminGenericTab({ title, icon, settings, setSettings, fields, onSave }: {
  1963→  title: string; icon: React.ReactNode;
  1964→  settings: Record<string, unknown>;
  1965→  setSettings: React.Dispatch<React.SetStateAction<Record<string, unknown>>>;
  1966→  fields: { key: string; label: string; description?: string; type: 'toggle' | 'select' | 'input' | 'textarea' | 'slider' | 'readonly'; options?: { value: string; label: string }[]; min?: number; max?: number; readOnlyValue?: string; }[];
  1967→  onSave: () => void;
  1968→}) {
  1969→  const update = (key: string, value: unknown) => {
  1970→    setSettings((prev) => ({ ...prev, [key]: value }));
  1971→  };
  1972→
  1973→  return (
  1974→    <div className="space-y-4">
  1975→      <SettingCard>
  1976→        <SectionHeader title={title} icon={icon} />
  1977→        {fields.map((f, i) => (
  1978→          <React.Fragment key={f.key}>
  1979→            {i > 0 && <Separator className="bg-white/5" />}
  1980→            {f.type === 'toggle' ? (
  1981→              <SettingRow label={f.label} description={f.description}>
  1982→                <Switch checked={!!settings[f.key]} onCheckedChange={(v) => update(f.key, v)} />
  1983→              </SettingRow>
  1984→            ) : f.type === 'select' ? (
  1985→              <SettingRow label={f.label} description={f.description}>
  1986→                <GlassSelect
  1987→                  value={String(settings[f.key] || '')}
  1988→                  onValueChange={(v) => update(f.key, v)}
  1989→                  options={f.options || []}
  1990→                />
  1991→              </SettingRow>
  1992→            ) : f.type === 'input' ? (
  1993→              <SettingRow label={f.label} description={f.description}>
  1994→                <GlassInput value={String(settings[f.key] || '')} onChange={(v) => update(f.key, v)} className="w-full sm:w-64" />
  1995→              </SettingRow>
  1996→            ) : f.type === 'textarea' ? (
  1997→              <SettingRow label={f.label} description={f.description}>
  1998→                <GlassTextarea value={String(settings[f.key] || '')} onChange={(v) => update(f.key, v)} rows={3} />
  1999→              </SettingRow>
  2000→            ) : f.type === 'slider' ? (
  2001→              <SettingRow label={`${f.label}: ${settings[f.key] || 0}`} description={f.description}>
  2002→                <div className="w-full sm:w-64">
  2003→                  <Slider value={[Number(settings[f.key] || 0)]} min={f.min || 0} max={f.max || 100} step={1} onValueChange={(v) => update(f.key, v[0])} />
  2004→                </div>
  2005→              </SettingRow>
  2006→            ) : f.type === 'readonly' ? (
  2007→              <SettingRow label={f.label}>
  2008→                <span className="text-sm text-[var(--muted-foreground)]">{f.readOnlyValue || String(settings[f.key] || 'N/A')}</span>
  2009→              </SettingRow>
  2010→            ) : null}
  2011→          </React.Fragment>
  2012→        ))}
  2013→      </SettingCard>
  2014→      <div className="flex justify-end">
  2015→        <SaveButton onClick={onSave} saving={false} saved={false} />
  2016→      </div>
  2017→    </div>
  2018→  );
  2019→}

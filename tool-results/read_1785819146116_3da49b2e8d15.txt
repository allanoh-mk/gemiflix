    26→    26→function getVolumeIcon(volume: number, isMuted: boolean): VolumeIcon {
    27→    27→  if (isMuted || volume === 0) return 'muted';

    32→    32→function formatTime(seconds: number): string {
    33→    33→  const h = Math.floor(seconds / 3600);
    34→    34→  const m = Math.floor((seconds % 3600) / 60);
    35→    35→  const s = Math.floor(seconds % 60);
    36→    36→  if (h > 0) {
    37→    37→    return `${h}:${m.toString().padStart(2, '0')}:${s.toString().padStart(2, '0')}`;
    38→    38→  }
    39→    39→  return `${m.toString().padStart(2, '0')}:${s.toString().padStart(2, '0')}`;
    40→    40→}

    42→    42→function formatRemainingTime(current: number, total: number): string {
    43→    43→  const remaining = Math.max(0, total - current);
    44→    44→  const negH = Math.floor(remaining / 3600);
    45→    45→  const negM = Math.floor((remaining % 3600) / 60);
    46→    46→  const negS = Math.floor(remaining % 60);
    47→    47→  const totH = Math.floor(total / 3600);
    48→    48→  const totM = Math.floor((total % 3600) / 60);
    49→    49→  const totS = Math.floor(total % 60);
    50→    50→  const fmtRemain = negH > 0
    51→    51→    ? `-${negH}:${negM.toString().padStart(2, '0')}:${negS.toString().padStart(2, '0')}`
    52→    52→    : `-${negM.toString().padStart(2, '0')}:${negS.toString().padStart(2, '0')}`;
    53→    53→  const fmtTotal = totH > 0
    54→    54→    ? `${totH}:${totM.toString().padStart(2, '0')}:${totS.toString().padStart(2, '0')}`
    55→    55→    : `${totM.toString().padStart(2, '0')}:${totS.toString().padStart(2, '0')}`;
    56→    56→  return `${fmtRemain} / ${fmtTotal}`;
    57→    57→}

    59→    59→const PROGRESS_STORAGE_KEY = 'gemiflix-player-progress';
    60→    60→
    61→    61→function loadSavedProgress(movieId: string): number {
    62→    62→  if (typeof window === 'undefined') return 0;
    63→    63→  try {
    64→    64→    const raw = localStorage.getItem(PROGRESS_STORAGE_KEY);
    65→    65→    if (!raw) return 0;
    66→    66→    const map = JSON.parse(raw) as Record<string, number>;
    67→    67→    return map[movieId] ?? 0;
    68→    68→  } catch {
    69→    69→    return 0;
    70→    70→  }
    71→    71→}
    72→    72→
    73→    73→function saveProgressToStorage(movieId: string, time: number): void {
    74→    74→  if (typeof window === 'undefined') return;
    75→    75→  try {
    76→    76→    const raw = localStorage.getItem(PROGRESS_STORAGE_KEY);
    77→    77→    const map = raw ? (JSON.parse(raw) as Record<string, number>) : {};
    78→    78→    map[movieId] = time;
    79→    79→    localStorage.setItem(PROGRESS_STORAGE_KEY, JSON.stringify(map));
    80→    80→  } catch {
    81→    81→    // storage full or unavailable
    82→    82→  }
    83→    83→}
